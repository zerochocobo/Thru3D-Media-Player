cmake_minimum_required(VERSION 3.22)
project(MvcDecoderProbe LANGUAGES C CXX)
include(CTest)
add_executable(mvc_recovery_gate_test recovery_gate_test.c)
set_target_properties(mvc_recovery_gate_test PROPERTIES C_STANDARD 11)
add_test(NAME mvc_recovery_gate COMMAND mvc_recovery_gate_test)
add_executable(mvc_config_test mvc_config_test.c)
set_target_properties(mvc_config_test PROPERTIES C_STANDARD 11)
add_test(NAME mvc_config COMMAND mvc_config_test)
file(TO_CMAKE_PATH "${EDGE264MVC_SOURCE}" EDGE264MVC_SOURCE)
if(NOT EXISTS "${EDGE264MVC_SOURCE}/edge264mvc.h")
    message(FATAL_ERROR "Set EDGE264MVC_SOURCE to the pinned edge264-mvc source checkout")
endif()
find_package(Threads REQUIRED)
add_library(edge264mvc SHARED "${EDGE264MVC_SOURCE}/src/edge264mvc.c")
target_include_directories(edge264mvc PUBLIC "${EDGE264MVC_SOURCE}")
target_compile_definitions(edge264mvc PRIVATE EDGE264MVC_BUILD _GNU_SOURCE)
target_compile_options(edge264mvc PRIVATE -O3 -fwrapv -flax-vector-conversions -Wno-override-init -fvisibility=hidden)
set_target_properties(edge264mvc PROPERTIES C_STANDARD 11 C_EXTENSIONS ON)
target_link_libraries(edge264mvc PRIVATE Threads::Threads)
add_executable(mvc_probe mvc_probe.cpp)
target_compile_features(mvc_probe PRIVATE cxx_std_17)
target_compile_options(mvc_probe PRIVATE -O3 -Wall -Wextra -Werror)
target_link_libraries(mvc_probe PRIVATE edge264mvc)
if(ANDROID)
    target_compile_options(edge264mvc PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/android_alloc_compat.h")
    target_link_options(edge264mvc PRIVATE "-Wl,-z,max-page-size=16384")
    target_link_options(mvc_probe PRIVATE "-Wl,-z,max-page-size=16384")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    target_compile_options(edge264mvc PRIVATE -msse4.1)
endif()
