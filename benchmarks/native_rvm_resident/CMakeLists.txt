cmake_minimum_required(VERSION 3.22)
project(RvmResidentOracle LANGUAGES CXX)
find_package(ncnn CONFIG REQUIRED)
add_executable(rvm_resident_oracle resident_oracle.cpp ../../native/rvm/rvm_backend.cpp ../../native/rvm/rvm_channel_mean.cpp ../../native/rvm/rvm_identity_tile.cpp ../../native/rvm/rvm_state_guard.cpp ../../native/rvm/rvm_resident_validation.cpp)
target_compile_features(rvm_resident_oracle PRIVATE cxx_std_17)
target_include_directories(rvm_resident_oracle PRIVATE ../../native/rvm ../../build/rvm/include)
target_link_libraries(rvm_resident_oracle PRIVATE ncnn)
add_executable(rvm_layer_precision layer_precision.cpp ../../native/rvm/rvm_backend.cpp ../../native/rvm/rvm_channel_mean.cpp ../../native/rvm/rvm_identity_tile.cpp ../../native/rvm/rvm_state_guard.cpp)
target_compile_features(rvm_layer_precision PRIVATE cxx_std_17)
target_include_directories(rvm_layer_precision PRIVATE ../../native/rvm ../../build/rvm/include)
target_link_libraries(rvm_layer_precision PRIVATE ncnn)
if(MSVC)
    target_compile_options(rvm_resident_oracle PRIVATE /EHsc /W4)
    target_compile_options(rvm_layer_precision PRIVATE /EHsc /W4)
endif()
if(ANDROID)
    # Standalone device oracle: same resident backend, no player/APK replacement.
    # ncnn exports -fno-exceptions; these validation sources catch failures.
    get_target_property(oracle_sources rvm_resident_oracle SOURCES)
    get_target_property(precision_sources rvm_layer_precision SOURCES)
    set_source_files_properties(${oracle_sources} ${precision_sources}
        PROPERTIES COMPILE_OPTIONS "-fexceptions;-fno-fast-math")
    foreach(target IN ITEMS rvm_resident_oracle rvm_layer_precision)
        target_link_libraries(${target} PRIVATE android log)
        target_link_options(${target} PRIVATE "-Wl,-z,max-page-size=16384")
    endforeach()
endif()
