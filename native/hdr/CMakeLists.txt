cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ThruHdr LANGUAGES C CXX)
endif()
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
# Imath otherwise writes a global _d cache default, changing later JNI targets
# as well as quest_hdr. Android loads the same explicit name in every build.
set(CMAKE_DEBUG_POSTFIX "" CACHE STRING "Keep Android JNI library names constant" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(IMATH_INSTALL OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PYTHON OFF CACHE BOOL "" FORCE)
FetchContent_Declare(Imath URL "${CMAKE_CURRENT_LIST_DIR}/../../third_party/openexr/imath-3.1.12.tar.gz"
    URL_HASH SHA256=8a1bc258f3149b5729c2f4f8ffd337c0e57f09096e4ba9784329f40c4a9035da DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(Imath)
set(LIBDEFLATE_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(LIBDEFLATE_BUILD_GZIP OFF CACHE BOOL "" FORCE)
set(LIBDEFLATE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ThruDeflate URL "${CMAKE_CURRENT_LIST_DIR}/../../third_party/openexr/libdeflate-1.24.tar.gz"
    URL_HASH SHA256=ad8d3723d0065c4723ab738be9723f2ff1cb0f1571e8bfcf0301ff9661f475e8 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(ThruDeflate)
set(EXR_DEFLATE_LIB libdeflate::libdeflate_static)
set(EXR_DEFLATE_VERSION 1.24)
set(OPENEXR_FORCE_INTERNAL_DEFLATE ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_IMATH ON CACHE BOOL "" FORCE)
foreach(option OPENEXR_INSTALL OPENEXR_BUILD_TOOLS OPENEXR_BUILD_EXAMPLES OPENEXR_BUILD_PYTHON OPENEXR_TEST_LIBRARIES OPENEXR_TEST_TOOLS)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(ThruExr URL "${CMAKE_CURRENT_LIST_DIR}/../../third_party/openexr/openexr-3.3.14.tar.gz"
    URL_HASH SHA256=54d931aaf50402c21ab7a0de9b390f91586d729a603455e1fe402dd3783426e0 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(ThruExr)
target_compile_options(OpenEXRCore PRIVATE "$<$<CONFIG:Debug>:-O2>")
add_library(quest_hdr SHARED hdr_image.cpp hdr_jni.cpp)
target_compile_features(quest_hdr PRIVATE cxx_std_17)
target_compile_options(quest_hdr PRIVATE -Wall -Wextra -Werror -fexceptions "$<$<CONFIG:Debug>:-O2>")
target_link_libraries(quest_hdr PRIVATE OpenEXR::OpenEXRCore)
target_link_options(quest_hdr PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
add_executable(hdr_probe EXCLUDE_FROM_ALL ../../tests/native/hdr_probe.cpp hdr_image.cpp)
target_compile_features(hdr_probe PRIVATE cxx_std_17)
target_link_libraries(hdr_probe PRIVATE OpenEXR::OpenEXRCore)
