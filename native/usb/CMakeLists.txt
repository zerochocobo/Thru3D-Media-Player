set(USB_DEPS "${CMAKE_CURRENT_LIST_DIR}/../../third_party/usb")
function(usb_write_if_changed path content)
    if(EXISTS "${path}")
        file(READ "${path}" previous)
        if(previous STREQUAL content)
            return()
        endif()
    endif()
    file(WRITE "${path}" "${content}")
endfunction()
function(usb_extract name hash sentinel)
    file(SHA256 "${USB_DEPS}/${name}" actual)
    if(NOT actual STREQUAL hash)
        message(FATAL_ERROR "USB dependency checksum mismatch: ${name}")
    endif()
    set(marker "${CMAKE_CURRENT_BINARY_DIR}/deps/${name}.ok")
    set(recorded "")
    if(EXISTS "${marker}")
        file(READ "${marker}" recorded)
    endif()
    if(NOT recorded STREQUAL hash OR NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/deps/${sentinel}")
        file(ARCHIVE_EXTRACT INPUT "${USB_DEPS}/${name}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/deps")
        if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/deps/${sentinel}")
            message(FATAL_ERROR "USB dependency extraction incomplete: ${name}")
        endif()
        file(WRITE "${marker}" "${hash}")
    endif()
endfunction()

foreach(header stdint.h inttypes.h stddef.h stdbool.h stdio.h stdlib.h stdarg.h string.h strings.h sys/types.h sys/stat.h sys/param.h sys/ioctl.h sys/mount.h sys/vfs.h sys/statvfs.h linux/fs.h linux/hdreg.h errno.h fcntl.h unistd.h time.h wchar.h locale.h limits.h ctype.h libgen.h endian.h byteswap.h malloc.h pwd.h mntent.h)
    string(TOUPPER "${header}" macro)
    string(REGEX REPLACE "[/.]" "_" macro "${macro}")
    check_include_file("${header}" HAVE_${macro})
    if(HAVE_${macro})
        string(APPEND nc "#define HAVE_${macro} 1\n")
    endif()
endforeach()
