cmake_minimum_required(VERSION 3.22)
project(Thru3DAirplay LANGUAGES C CXX)
option(AIRPLAY_HOST_TESTS "Build the host-only pairing regression server" OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(NOT AIRPLAY_DEPS)
    set(AIRPLAY_DEPS "${CMAKE_CURRENT_LIST_DIR}/../../build/airplay/deps")
endif()
set(UX "${AIRPLAY_DEPS}/uxplay/lib")
set(PL "${AIRPLAY_DEPS}/libplist")
set(CRYPTO "${AIRPLAY_DEPS}/crypto")
if(NOT EXISTS "${CRYPTO}/lib/libcrypto.a" OR NOT EXISTS "${UX}/raop.c")
    message(FATAL_ERROR "Run tools/environment/Prepare-Airplay.ps1 to restore the pinned AirPlay sources and crypto library")
endif()
add_library(airplay_crypto UNKNOWN IMPORTED)
if(AIRPLAY_HOST_TESTS)
    find_file(HOST_CRYPTO_LIBRARY NAMES libcrypto.so.3 PATHS /usr/lib/x86_64-linux-gnu /lib/x86_64-linux-gnu REQUIRED)
    set_target_properties(airplay_crypto PROPERTIES IMPORTED_LOCATION "${HOST_CRYPTO_LIBRARY}")
else()
    set_target_properties(airplay_crypto PROPERTIES IMPORTED_LOCATION "${CRYPTO}/lib/libcrypto.a")
endif()
file(GLOB plist_sources "${PL}/src/*.c")
list(FILTER plist_sources EXCLUDE REGEX "/(plistutil|plutil)\\.c$")
add_library(airplay_plist STATIC ${plist_sources} "${PL}/libcnary/cnary.c" "${PL}/libcnary/node.c" "${PL}/libcnary/node_list.c")
target_include_directories(airplay_plist PUBLIC "${PL}/include" PRIVATE "${PL}/src" "${PL}/libcnary/include")
target_compile_definitions(airplay_plist PRIVATE _GNU_SOURCE HAVE_STRNDUP PACKAGE_VERSION="2.6.0")
set(core airplay_video byteutils compat crypto fairplay_playfair http_request http_response httpd logger mirror_buffer netutils pairing raop raop_buffer raop_ntp raop_rtp raop_rtp_mirror srp utils)
set(protocol_sources)
foreach(name IN LISTS core)
    list(APPEND protocol_sources "${UX}/${name}.c")
endforeach()
if(AIRPLAY_HOST_TESTS)
    add_executable(airplay_pairing_server tests/pairing_server.c android_dnssd_shim.c ${protocol_sources}
        "${UX}/llhttp/api.c" "${UX}/llhttp/http.c" "${UX}/llhttp/llhttp.c"
        "${UX}/playfair/hand_garble.c" "${UX}/playfair/modified_md5.c" "${UX}/playfair/omg_hax.c"
        "${UX}/playfair/playfair.c" "${UX}/playfair/sap_hash.c")
    target_include_directories(airplay_pairing_server PRIVATE "${UX}" "${UX}/llhttp" "${UX}/playfair" "${CRYPTO}/include")
    target_compile_definitions(airplay_pairing_server PRIVATE PLIST_210 TARGET_RT_LITTLE_ENDIAN=1)
    target_link_libraries(airplay_pairing_server PRIVATE airplay_plist airplay_crypto pthread m)
    return()
endif()
add_library(quest_airplay SHARED airplay_jni.cpp android_dnssd_shim.c ${protocol_sources}
    "${UX}/llhttp/api.c" "${UX}/llhttp/http.c" "${UX}/llhttp/llhttp.c"
    "${UX}/playfair/hand_garble.c" "${UX}/playfair/modified_md5.c" "${UX}/playfair/omg_hax.c"
    "${UX}/playfair/playfair.c" "${UX}/playfair/sap_hash.c")
target_include_directories(quest_airplay PRIVATE "${UX}" "${UX}/llhttp" "${UX}/playfair" "${CRYPTO}/include")
target_compile_definitions(quest_airplay PRIVATE ANDROID PLIST_210 TARGET_RT_LITTLE_ENDIAN=1)
target_link_libraries(quest_airplay PRIVATE airplay_plist airplay_crypto android log m)
target_link_options(quest_airplay PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
