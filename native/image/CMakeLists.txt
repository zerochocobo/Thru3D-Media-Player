set(IMAGE_DEPS "${CMAKE_CURRENT_LIST_DIR}/../../third_party/image")
function(image_source name archive hash)
    FetchContent_Declare(${name} URL "${IMAGE_DEPS}/${archive}" URL_HASH SHA256=${hash}
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR _source_only)
    FetchContent_MakeAvailable(${name})
    set(${name}_SOURCE_DIR "${${name}_SOURCE_DIR}" PARENT_SCOPE)
    set(${name}_BINARY_DIR "${${name}_BINARY_DIR}" PARENT_SCOPE)
endfunction()
