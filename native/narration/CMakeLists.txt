set(NARRATION_DEPS "${CMAKE_CURRENT_LIST_DIR}/../../tmp/narration-deps")
if(NOT EXISTS "${NARRATION_DEPS}/libonnxruntime.so")
    message(FATAL_ERROR "Run tools/environment/Prepare-Narration.ps1")
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_DISABLE_EMBEDDED_DATA ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${NARRATION_DEPS}/sentencepiece-0.2.1" "sentencepiece" EXCLUDE_FROM_ALL)
target_link_libraries(sentencepiece-static INTERFACE log)
add_library(narration_ort SHARED IMPORTED)
set_target_properties(narration_ort PROPERTIES IMPORTED_LOCATION "${NARRATION_DEPS}/libonnxruntime.so")
add_library(narration_avformat SHARED IMPORTED)
set_target_properties(narration_avformat PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../../android/player-plugin/src/main/jniLibs/arm64-v8a/libavformat.so")
add_library(narration_avutil SHARED IMPORTED)
set_target_properties(narration_avutil PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../../android/player-plugin/src/main/jniLibs/arm64-v8a/libavutil.so")
add_library(narration_avcodec SHARED IMPORTED)
set_target_properties(narration_avcodec PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../../android/player-plugin/src/main/jniLibs/arm64-v8a/libavcodec.so")
add_library(narration_swresample SHARED IMPORTED)
set_target_properties(narration_swresample PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../../android/player-plugin/src/main/jniLibs/arm64-v8a/libswresample.so")
add_library(quest_narration SHARED narration_jni.cpp subtitle_reader.cpp voice_encoder.cpp voice_segment.cpp)
target_compile_features(quest_narration PRIVATE cxx_std_17)
target_compile_options(quest_narration PRIVATE -O2 -fexceptions -fno-fast-math -Wall -Wextra -Werror)
target_include_directories(quest_narration SYSTEM PRIVATE "${NARRATION_DEPS}" "${NARRATION_DEPS}/sentencepiece-0.2.1/src"
    "${NARRATION_DEPS}/FFmpeg-094a2f8a2a5e7fa64736e067de224ce28fdf5979")
target_link_libraries(quest_narration PRIVATE narration_ort sentencepiece-static narration_avformat narration_avutil narration_avcodec narration_swresample log android)
target_link_options(quest_narration PRIVATE "-Wl,-z,max-page-size=16384")
